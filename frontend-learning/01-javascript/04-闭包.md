# JavaScript 闭包

## 一句话总结

闭包是一个函数，它能记住并访问定义时的词法作用域，即使函数在该作用域之外执行。

---

## 核心概念

### 1. 什么是闭包？

```javascript
function outer() {
    let count = 0;
    function inner() {
        count++;
        console.log(count);
    }
    return inner;
}

let counter = outer();
counter();  // 1
counter();  // 2
counter();  // 3
```

**发生了什么？**

1. `outer()` 执行完毕，按理说 `count` 应该被销毁
2. 但是 `inner` 函数引用了 `count`
3. 所以 `count` 被保留下来，`inner` 形成了闭包

### 2. 闭包的本质

**闭包 = 函数 + 其词法环境的引用**

```
outer() 的作用域
┌──────────────────────┐
│ count = 0            │ ← 这个变量被 inner "捕获"了
│                      │
│ inner (函数)         │
│   ↓                  │
│   引用 count         │
└──────────────────────┘

outer() 执行完后：
┌──────────────────────┐
│ count = 0            │ ← 没有被销毁，因为 inner 还在引用
└──────────────────────┘
```

### 3. 为什么需要闭包？

**问题：函数执行完，局部变量会被销毁**

```javascript
function counter() {
    let count = 0;
    count++;
    return count;
}

counter();  // 1
counter();  // 1 ← 每次调用都是新的 count
```

**解决方案：闭包让变量"活"下来**

```javascript
function counter() {
    let count = 0;
    return function() {
        count++;
        return count;
    };
}

let c = counter();
c();  // 1
c();  // 2
c();  // 3
```

## 闭包的使用场景

### 1. 数据私有化（封装）

```javascript
function createBankAccount(initialBalance) {
    let balance = initialBalance;  // 私有变量
    
    return {
        deposit(amount) {
            balance += amount;
            return balance;
        },
        withdraw(amount) {
            if (amount > balance) {
                return "Insufficient funds";
            }
            balance -= amount;
            return balance;
        },
        getBalance() {
            return balance;
        }
    };
}

let account = createBankAccount(1000);
account.deposit(500);      // 1500
account.withdraw(200);     // 1300
account.getBalance();      // 1300
// account.balance          // undefined ← 无法直接访问
```

### 2. 函数工厂

```javascript
function createMultiplier(multiplier) {
    return function(number) {
        return number * multiplier;
    };
}

let double = createMultiplier(2);
let triple = createMultiplier(3);

double(5);   // 10
triple(5);   // 15
```

### 3. 回调函数中保持状态

```javascript
function setupButton() {
    let clickCount = 0;
    
    return function() {
        clickCount++;
        console.log(`Button clicked ${clickCount} times`);
    };
}

let handleClick = setupButton();
handleClick();  // "Button clicked 1 times"
handleClick();  // "Button clicked 2 times"
```

### 4. 柯里化（Currying）

```javascript
function add(a) {
    return function(b) {
        return a + b;
    };
}

let add5 = add(5);
add5(3);   // 8
add5(10);  // 15
```

### 5. 防抖和节流

```javascript
// 防抖
function debounce(fn, delay) {
    let timer = null;
    return function(...args) {
        clearTimeout(timer);
        timer = setTimeout(() => {
            fn.apply(this, args);
        }, delay);
    };
}

// 节流
function throttle(fn, interval) {
    let lastTime = 0;
    return function(...args) {
        let now = Date.now();
        if (now - lastTime >= interval) {
            lastTime = now;
            fn.apply(this, args);
        }
    };
}
```

## 闭包的内存问题

### 闭包会导致内存泄漏吗？

```javascript
function createClosure() {
    let largeData = new Array(1000000).fill('x');
    return function() {
        // 不使用 largeData
        return "done";
    };
}

let closure = createClosure();
// largeData 被垃圾回收了吗？
```

**答案：取决于闭包是否引用了变量。**

- 如果闭包引用了 `largeData` → 不会被回收
- 如果闭包没有引用 `largeData` → 会被回收（V8 的优化）

### 如何避免内存泄漏？

```javascript
// ❌ 不好：闭包引用了不需要的变量
function createHeavyClosure() {
    let largeData = new Array(1000000).fill('x');
    let unusedData = new Array(1000000).fill('y');
    
    return function() {
        return largeData.length;  // 只用 largeData
    };
    // unusedData 不会被回收，因为闭包捕获了整个作用域
}

// ✅ 好：避免捕获不需要的变量
function createHeavyClosure() {
    let largeData = new Array(1000000).fill('x');
    
    return function() {
        return largeData.length;
    };
}
```

## 动手验证

```javascript
// 测试 1：基本闭包
function outer() {
    let count = 0;
    return function inner() {
        count++;
        return count;
    };
}

let counter = outer();
console.log(counter());  // 1
console.log(counter());  // 2
console.log(counter());  // 3

// 测试 2：循环中的闭包（经典问题）
for (var i = 0; i < 3; i++) {
    setTimeout(function() {
        console.log(i);
    }, 1000);
}
// 输出：3, 3, 3（不是 0, 1, 2）

// 解决方案 1：使用 let
for (let i = 0; i < 3; i++) {
    setTimeout(function() {
        console.log(i);
    }, 1000);
}
// 输出：0, 1, 2

// 解决方案 2：使用 IIFE 创建闭包
for (var i = 0; i < 3; i++) {
    (function(j) {
        setTimeout(function() {
            console.log(j);
        }, 1000);
    })(i);
}
// 输出：0, 1, 2

// 测试 3：闭包与 this
function Person(name) {
    this.name = name;
    this.sayHello = function() {
        setTimeout(function() {
            console.log(`Hello, I'm ${this.name}`);
        }, 1000);
    };
}

let peng = new Person("Peng");
peng.sayHello();  // "Hello, undefined" ← this 指向 window

// 解决方案：使用箭头函数
function Person(name) {
    this.name = name;
    this.sayHello = function() {
        setTimeout(() => {
            console.log(`Hello, I'm ${this.name}`);
        }, 1000);
    };
}
```

## 容易混淆的地方

1. **闭包不是"函数返回函数"** — 闭包是函数 + 词法环境的引用
2. **闭包会导致内存泄漏吗？** — 取决于是否引用了不需要的变量
3. **循环中的闭包问题** — 使用 `let` 或 IIFE 解决
4. **闭包中的 this** — 箭头函数继承外层 this，普通函数 this 指向 window

## 实际开发中的应用

```javascript
// Vue 2 的 data 函数就是闭包
export default {
    data() {
        return {
            name: "Peng"  // 每个组件实例都有独立的 data
        };
    }
};

// Vue 3 组合式 API 的 ref 也是闭包
import { ref } from 'vue';

function useCounter() {
    const count = ref(0);
    function increment() {
        count.value++;
    }
    return { count, increment };
}

// React Hooks 也是闭包
function useCounter() {
    const [count, setCount] = useState(0);
    const increment = () => setCount(count + 1);
    return { count, increment };
}
```
